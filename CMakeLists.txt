# ---------------------------------------------------------------- RNNoise
# Upstream has no CMake support; build the portable (non-RTCD) C path.
# Weights are compiled in (src/rnnoise_data.c), so rnnoise_create(NULL) works.
set(RNN ${CMAKE_SOURCE_DIR}/third_party/rnnoise)
add_library(rnnoise STATIC
  ${RNN}/src/denoise.c
  ${RNN}/src/rnn.c
  ${RNN}/src/pitch.c
  ${RNN}/src/kiss_fft.c
  ${RNN}/src/celt_lpc.c
  ${RNN}/src/nnet.c
  ${RNN}/src/nnet_default.c
  ${RNN}/src/parse_lpcnet_weights.c
  ${RNN}/src/rnnoise_data.c
  ${RNN}/src/rnnoise_tables.c)
target_include_directories(rnnoise PUBLIC ${RNN}/include PRIVATE ${RNN}/src)
target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD DISABLE_DEBUG_FLOAT)   # int8 weights only: 3-4x faster, 8 MB smaller
target_compile_options(rnnoise PRIVATE -O3 -w)
if(UNIX)
  target_link_libraries(rnnoise PUBLIC m)
endif()

# x86-64: RNNoise's own run-time CPU dispatch (SSE4.1 / AVX2+FMA kernels for the network layers);
# the portable C path stays as the fallback and is chosen on other CPUs. Opt out: -DVF_RNNOISE_SIMD=OFF.
option(VF_RNNOISE_SIMD "Use RNNoise's SSE4.1/AVX2 kernels with run-time dispatch (x86-64)" ON)
if(VF_RNNOISE_SIMD AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|x86_64|AMD64" AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_sources(rnnoise PRIVATE ${RNN}/src/x86/x86cpu.c ${RNN}/src/x86/x86_dnn_map.c
                                 ${RNN}/src/x86/nnet_sse4_1.c ${RNN}/src/x86/nnet_avx2.c)
  set_source_files_properties(${RNN}/src/x86/nnet_sse4_1.c PROPERTIES COMPILE_OPTIONS "-msse4.1")
  set_source_files_properties(${RNN}/src/x86/nnet_avx2.c PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
  target_compile_definitions(rnnoise PRIVATE RNN_ENABLE_X86_RTCD OPUS_HAVE_RTCD OPUS_X86_MAY_HAVE_SSE4_1
                             OPUS_X86_MAY_HAVE_AVX2 CPU_INFO_BY_C)
  target_include_directories(rnnoise PRIVATE ${RNN}/src/x86)
  message(STATUS "mic thingy: RNNoise SIMD dispatch enabled")
endif()

# Virtual microphone: PipeWire on Linux, stub elsewhere (Windows uses VB-Cable).
option(VF_PIPEWIRE "Build the PipeWire virtual microphone (Linux)" ${UNIX})
if(VF_PIPEWIRE AND UNIX AND NOT APPLE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
  target_sources(vfcore PRIVATE src/audio/vmic_pipewire.c)
  target_link_libraries(vfcore PUBLIC PkgConfig::PIPEWIRE)
  # SPA headers use GNU extensions that -Wpedantic flags.
  set_source_files_properties(src/audio/vmic_pipewire.c PROPERTIES COMPILE_OPTIONS "-Wno-pedantic")
  message(STATUS "mic thingy: PipeWire virtual microphone enabled")
else()
  target_sources(vfcore PRIVATE src/audio/vmic_stub.c)
endif()
if(MSVC)
  target_compile_options(vfcore PRIVATE /W4)
else()
  target_compile_options(vfcore PRIVATE -Wall -Wextra -Wpedantic)
endif()

# ---------------------------------------------------------------- executables
# Windows resources: the icon (assets/mic-thingy.ico, generated by tools/make_icon.c)
set(VF_RESOURCES "")
if(WIN32)
  enable_language(RC)
  set(VF_ICON_PATH "${CMAKE_SOURCE_DIR}/assets/mic-thingy.ico")
  configure_file(assets/mic-thingy.rc.in ${CMAKE_BINARY_DIR}/mic-thingy.rc @ONLY)
  # Own object library: windres breaks on the target include paths when the project
  # path contains spaces ("mic thingy"), and the resource needs none.
  add_library(vf_resources OBJECT ${CMAKE_BINARY_DIR}/mic-thingy.rc)
  # the .rc only names the .ico, so rebuild the resource whenever the icon file changes
  set_source_files_properties(${CMAKE_BINARY_DIR}/mic-thingy.rc PROPERTIES OBJECT_DEPENDS ${VF_ICON_PATH})
  set(VF_RESOURCES $<TARGET_OBJECTS:vf_resources>)
endif()

endif()

# ---------------------------------------------------------------- GUI (raylib + raygui)
option(VF_GUI "Build the raylib GUI (mic-thingy)" ON)
if(VF_GUI)
  # raylib without its audio module: raudio embeds its own miniaudio, which would
  # clash with ours, and mic thingy does all audio itself.
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
  set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
  set(SUPPORT_MODULE_RAUDIO OFF CACHE BOOL "" FORCE)
  set(USE_AUDIO OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/raylib EXCLUDE_FROM_ALL)
  if(TARGET raylib)
    target_compile_options(raylib PRIVATE -w)
  endif()
  add_executable(mic-thingy src/main_gui.c src/app/tray.c src/app/win_chrome.c ${VF_RESOURCES})
  target_include_directories(mic-thingy PRIVATE third_party/raygui)
  target_link_libraries(mic-thingy PRIVATE vfcore raylib)
  if(WIN32)
    target_link_libraries(mic-thingy PRIVATE shell32 user32 gdi32 dwmapi)
    set_target_properties(mic-thingy PROPERTIES WIN32_EXECUTABLE ON)   # no console window
  endif()
  if(NOT MSVC)
    # raygui itself triggers a few warnings; keep our own code strict elsewhere.
    target_compile_options(mic-thingy PRIVATE -Wall -Wno-unused-function -Wno-unused-parameter
                                               -Wno-unused-but-set-variable)
  endif()
endif()
